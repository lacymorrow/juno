fn format_agent_descriptions(&self) -> String {
    let mut descriptions = String::new();
    for name in self.get_agent_names() {
        descriptions.push_str(&format!("- {}: {}", name, self.get_agent_description(&name)));
    }
    descriptions
}

fn get_agent_description(&self, name: &str) -> String {
    match name {

        "UserInteractionAgent" => "Interacts with the user to ask questions or request information.",
        "InformationRetrievalAgent" => "Searches the web and reads documents to retrieve information.",
        _ => "Unknown agent type."
    }.to_string()
}

fn execute_step(&mut self, step: &mut PlanStep) -> Result<(), Error> {
    step.status = PlanStepStatus::InProgress;
    
    // Determine which agent to use
    let agent_name = step.agent_assigned.clone().unwrap_or_else(|| {
        // Use LLM to determine the best agent if none specified
        self.determine_best_agent_for_step(step)
    });
    
    let agent = self.agents.get_mut(&agent_name).ok_or(Error::AgentNotFound)?;
    
    // Execute the step
    match agent.execute_task(&step.description) {
        Ok(_) => {
            step.status = PlanStepStatus::Completed;
            Ok(())
        },
        Err(e) => {
            step.status = PlanStepStatus::Failed(e.to_string());
            Err(e)
        }
    }
}
