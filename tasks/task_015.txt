impl MacOSInteractionAgent {
    // ... existing methods ...
    
    fn get_hybrid_perception(&self) -> Result<HybridPerception, Error> {
        let screenshot = self.get_current_screenshot()?;
        let accessibility_tree = self.get_accessibility_tree()?;
        
        Ok(HybridPerception {
            screenshot,
            accessibility_tree
        })
    }
    
    fn execute_task(&mut self, task: &str) -> Result<String, Error> {
        // Get initial perception
        let perception = self.get_hybrid_perception()?;
        
        // Add to agent memory
        self.tool_call_agent.add_to_memory(Message::System {
            content: format!("""
            Current Screenshot (base64 encoded):
            {}
            
            Accessibility Tree:
            {}
            """, perception.screenshot, perception.accessibility_tree)
        });
        
        // Execute task using tool_call_agent
        self.tool_call_agent.execute_task(task)
    }
}
