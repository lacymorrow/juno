3. Create Tauri commands for running tests:
```rust
#[tauri::command]
async fn run_test(test_case: TestCase, state: tauri::State<'_, AppState>) -> Result<TestResult, String> {
    let plan_manager = Arc::clone(&state.plan_manager);
    let logger = Arc::clone(&state.logger);
    
    let test_runner = TestRunner::new(plan_manager, logger);
    Ok(test_runner.run_test(test_case).await)
}

#[tauri::command]
async fn run_test_suite(test_suite: Vec<TestCase>, state: tauri::State<'_, AppState>) -> Result<Vec<TestResult>, String> {
    let plan_manager = Arc::clone(&state.plan_manager);
    let logger = Arc::clone(&state.logger);
    
    let test_runner = TestRunner::new(plan_manager, logger);
    
    let mut results = Vec::new();
    for test_case in test_suite {
        results.push(test_runner.run_test(test_case).await);
    }
    
    Ok(results)
}
```
