```rust
        let log_dir = dirs::data_dir()
            .ok_or_else(|| Error::DataDirNotFound)?
            .join("dotdot")
            .join("logs");
        
        std::fs::create_dir_all(&log_dir)?;
        
        let log_file_name = format!("{}.log", Utc::now().format("%Y-%m-%d"));
        let log_path = log_dir.join(log_file_name);
        
        Ok(Self {
            log_path,
            min_level,
            buffer: Vec::new(),
            buffer_size: 100
        })
    }
    
    fn log(&mut self, level: LogLevel, component: &str, message: &str, details: Option<serde_json::Value>, session_id: Option<Uuid>) -> Result<(), Error> {
        if level < self.min_level {
            return Ok(());
        }
        
        let entry = LogEntry {
            timestamp: Utc::now(),
            level,
            component: component.to_string(),
            message: message.to_string(),
            details,
            session_id
        };
        
        self.buffer.push(entry.clone());
        
        if self.buffer.len() >= self.buffer_size {
            self.flush()?;
        }
        
        Ok(())
    }
```

4. Create a macro for easy logging:
```rust
macro_rules! log {
    ($app_handle:expr, $level:expr, $component:expr, $message:expr) => {
        if let Some(state) = $app_handle.state::<AppState>() {
            if let Ok(mut logger) = state.logger.try_lock() {
                let _ = logger.log($level, $component, $message, None, None);
            }
        }
    };
    ($app_handle:expr, $level:expr, $component:expr, $message:expr, $details:expr) => {
        if let Some(state) = $app_handle.state::<AppState>() {
            if let Ok(mut logger) = state.logger.try_lock() {
                let _ = logger.log($level, $component, $message, Some($details), None);
            }
        }
    };
    ($app_handle:expr, $level:expr, $component:expr, $message:expr, $details:expr, $session_id:expr) => {
        if let Some(state) = $app_handle.state::<AppState>() {
            if let Ok(mut logger) = state.logger.try_lock() {
                let _ = logger.log($level, $component, $message, Some($details), Some($session_id));
            }
        }
    };
}
```

5. Create Tauri commands for accessing logs:
```rust
#[tauri::command]
async fn get_logs(level: Option<String>, component: Option<String>, session_id: Option<String>, limit: Option<usize>, state: tauri::State<'_, AppState>) -> Result<Vec<LogEntry>, String> {
    let logger = state.logger.lock().await;
    
    let log_level = level.map(|l| match l.as_str() {
        "debug" => LogLevel::Debug,
        "info" => LogLevel::Info,
        "warning" => LogLevel::Warning,
        "error" => LogLevel::Error,
        _ => LogLevel::Info
    });
    
    let session_uuid = if let Some(id) = session_id {
        Some(Uuid::parse_str(&id).map_err(|e| e.to_string())?)
    } else {
        None
    };
    
    logger.get_logs(log_level, component.as_deref(), session_uuid, limit.unwrap_or(100))
        .map_err(|e| e.to_string())
}
```
