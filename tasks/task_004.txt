3. Create a factory function for creating LLM interfaces based on configuration:
```rust
fn create_llm_interface(config: &LLMConfig) -> Box<dyn LLMInterface> {
    match config.provider.as_str() {
        "anthropic" => Box::new(AnthropicInterface::new(config.api_key.clone(), config.model.clone())),
        "openai" => Box::new(OpenAIInterface::new(config.api_key.clone(), config.model.clone())),
        _ => panic!("Unsupported LLM provider: {}", config.provider)
    }
}
```
