2. Implement the `ConfigurationManager` struct:
```rust
struct ConfigurationManager {
    config: Configuration,
    config_path: PathBuf
}

impl ConfigurationManager {
    fn new() -> Result<Self, Error> {
        let config_dir = dirs::config_dir()
            .ok_or_else(|| Error::ConfigDirNotFound)?
            .join("dotdot");
        
        std::fs::create_dir_all(&config_dir)?;
        
        let config_path = config_dir.join("config.json");
        
        let config = if config_path.exists() {
            let config_str = std::fs::read_to_string(&config_path)?;
            serde_json::from_str(&config_str)?
        } else {
            let default_config = Configuration::default();
            let config_str = serde_json::to_string_pretty(&default_config)?;
            std::fs::write(&config_path, config_str)?;
            default_config
        };
        
        Ok(Self {
            config,
            config_path
        })
    }
    
    fn get_config(&self) -> &Configuration {
        &self.config
    }
    
    fn update_config(&mut self, config: Configuration) -> Result<(), Error> {
        self.config = config;
        self.save_config()
    }
    
    fn save_config(&self) -> Result<(), Error> {
        let config_str = serde_json::to_string_pretty(&self.config)?;
        std::fs::write(&self.config_path, config_str)?;
        Ok(())
    }
}

impl Default for Configuration {
```

```rust
#[tauri::command]
async fn get_config(state: tauri::State<'_, AppState>) -> Result<Configuration, String> {
    let manager = state.config_manager.lock().await;
    Ok(manager.get_config().clone())
}

#[tauri::command]
async fn update_config(config: Configuration, state: tauri::State<'_, AppState>) -> Result<(), String> {
    let mut manager = state.config_manager.lock().await;
    manager.update_config(config)
        .map_err(|e| e.to_string())
}
```
