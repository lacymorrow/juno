impl ToolCallAgent {
    // ... existing methods ...
    
    fn execute_tool_call_with_recovery(&self, tool_call: &ToolCall, max_retries: usize) -> Result<ToolResult, Error> {
        let mut retries = 0;
        let mut last_error = None;
        
        while retries < max_retries {
            match self.execute_tool_call(tool_call) {
                Ok(result) => return Ok(result),
                Err(e) => {
                    last_error = Some(e.clone());
                    
                    // Determine error pattern
                    let pattern = self.determine_error_pattern(&e);
                    
                    // Choose recovery strategy
                    let strategy = self.choose_recovery_strategy(pattern, retries);
                    
                    // Apply recovery strategy
                    match self.apply_recovery_strategy(strategy, tool_call, &e) {
                        Ok(new_tool_call) => {
                            // Try with the new tool call
                            match self.execute_tool_call(&new_tool_call) {
                                Ok(result) => return Ok(result),
                                Err(e) => last_error = Some(e)
                            }
                        },
                        Err(_) => {} // Couldn't apply recovery strategy
                    }
                    
                    retries += 1;
                }
            }
        }
        
        Err(last_error.unwrap_or_else(|| Error::MaxRetriesExceeded))
    }
